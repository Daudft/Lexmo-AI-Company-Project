How it works

From idea to production
in three moves.

compose.py 
from evolve import Agent, tools, memory

agent = Agent(
    model="evolve-reason-2",
    memory=memory.vector("support-kb"),
    tools=[tools.http, tools.sql("orders")],
    guardrails=["pii", "policy:refunds"],
)

reply = agent.run("Where is order #4182?")
print(reply.text)
from evolve import evals

report = evals.replay(
    agent,
    dataset="prod-traffic-7d",
    metrics=["accuracy", "latency_p95", "cost"],
)

report.compare(baseline="v41")
# accuracy  +4.2%   latency_p95  -38ms   cost  -12%
$ evolve deploy support-agent --regions global
